\begin{question}
    \item {}
    จงหาผลเฉลยในรูปอนุกรมกำลังรอบจุด $x=0$ ของสมการ $2x^2y'' + x(2x+1)y' - y = 0$ โดยวิธีของโฟรเบนิอุส \qmark{12}

    % ----------------------------------------
    \bigskip
    Let $P(x) = (2 + \frac1x)$ and $Q(x) = \frac1{x^2}$.
    Then, $x = 0$ is a regular singular point because $xP(x)$ and $x^2Q(x)$ are analytic functions at $x = 0$.

    \medskip
    Suppose that $\displaystyle y = \sum_{n=0}^\infty a_n x^{n+r}, \quad a_n \neq 0$
    \begin{align*}
        2 x^2 \sum_{n=0}^\infty (n+r)(n+r-1) a_n x^{n+r-2} + x(2x + 1) \sum_{n=0}^\infty (n+r) a_n x^{n+r-1} - \sum_{n=0}^\infty a_n x^{n+r} & = 0 \\
        \sum_{n=0}^\infty \left[ 2 (n+r)(n+r-1) a_n x^{n+r} + 2 (n+r) a_n x^{n+r+1} + (n+r) a_n x^{n+r} - a_n x^{n+r} \right]                & = 0 \\
    \end{align*}

    \vspace{-2em}

    The coefficient of the lowest degree term $x^r$ is
    $$
        (2r(r-1) + r - 1)a_n
    $$

    Solve the indicial equation
    \begin{align*}
        (2r(r-1) + r - 1)a_n & = 0                                           \\
        2r^2 - r - 1         & = 0                                           \\
        r                    & = \frac{1 \pm \sqrt{(-1)^2 - 4(2)(-1)}}{2(2)} \\
                             & = \frac{1 \pm 3}{4}                           \\
                             & = -\frac12, 1
    \end{align*}

    Find the recursive formula (set powers of $x$ to $n+r+1$)
    $$
        2 (n+r+1)(n+r) a_{n+1} + 2 (n+r) a_n + (n+r+1) a_{n+1} - a_{n+1} = 0                                                              \\
    $$
    \vspace*{-2em}
    \begin{align*}
        a_{n+1} & = -\frac{2(n+r) a_n}{2(n+r+1)(n+r) + (n+r)} \\
                & = - \frac{2 a_n}{2(n+r+1) + 1}
    \end{align*}

    Case $r_1 = -\frac12$
    \begin{align*}
        a_{n+1} & = - \frac{2 a_n}{2n - 1 + 2 + 1}                    \\
                & = - \frac{a_n}{n + 1}                               \\
        a_n     & = \frac{(-1)^n}{n!} a_0                             \\
        \intertext{Let $c_1 = a_0$}
        y_1     & = c_1 \sum_{n=0}^\infty \frac{(-1)^n}{n!} x^{n-1/2} \\
                & = c_1 \frac{e^{-x}}{\sqrt{x}}
    \end{align*}

    Case $r_2 = 1$
    \begin{align*}
        a_{n+1} & = - \frac{2 a_n}{2n + 5} \\
    \end{align*}

    \vspace{-2em}
    Let $c_2 = a_0$
    \vspace{1em}
    $$
        y_2 = c_2 \left(x - \frac25 x^2 + \frac{4}{7\cdot5} x^3 - \frac{8}{9\cdot7\cdot5} x^4 + \dots\right)
    $$

    \medskip

    The general solution is
    $$
        \boxed{y = c_1 \frac{e^{-x}}{\sqrt{x}} + c_2 \left(x - \frac25 x^2 + \frac{4}{7\cdot5} x^3 - \frac{8}{9\cdot7\cdot5} x^4 + \dots\right)}
    $$
    % ----------------------------------------

    \item {}
    จงหาค่าของ
    \begin{question}
        \item
        $\displaystyle \int_0^\infty e^{-x^4} \dd{x} \quad \text{(ให้ $t = x^4$)}$ \qmark{3}

        % ----------------------------------------
        \begin{align*}
            \int_0^\infty \frac14 e^{-t} t^{-3/4} \dd{t} & = \frac14 \mathcal{L}\left\{t^{-3/4}\right\}(1) \\
                                                         & = \boxed{\frac14 \Gamma\left(\frac14\right)}
        \end{align*}
        % ----------------------------------------

        \pagebreak

        \item
        $\displaystyle \int_{2\pi}^\infty e^{-4t} \sin 2t \cos 2t \dd{t}$ \qmark{4}

        % ----------------------------------------
        \begin{align*}
            \int_0^\infty \frac12 H(t-2\pi) e^{-4t} \sin 4t \dd{t} & = \frac12 \mathcal{L}\left\{H(t-2\pi) \sin 4t \right\}(4)                \\
                                                                   & = \frac14 e^{-2\pi \cdot 4} \mathcal{L}\left\{\sin(4t + 2\pi)\right\}(4) \\
                                                                   & = \boxed{\frac{e^{-8\pi}}{32}}
        \end{align*}
        % ----------------------------------------

    \end{question}

    \item {}
    จงเขียนสมการเชิงอนุพันธ์ที่มี $P_4(x)$ เป็นผลเฉลย พร้อมทั้งหา $P_4(x)$ \qmark{3}

    % ----------------------------------------
    \bigskip
    Legendre's differential equation of order $4$ is
    $$
        \boxed{(1-x^2)y'' - 2xy' + 20y = 0}
    $$

    Use Rodrigues' Formula
    \begin{align*}
        P_4(x) & = \frac{1}{2^n n!} \dv[n]{x} (x^2 - 1)^n \bigg|_{n=4}                                   \\
               & = \frac{1}{2^4 4!} \dv[4]{x} (x^2 - 1)^4                                                \\
               & = \frac{1}{2^4 4!} \dv[4]{x} (x^8 - 4x^6 + 6x^4)                                        \\
               & = \frac{1}{2^4 4!} (8\cdot7\cdot6\cdot5 x^4 - 4\cdot6\cdot5\cdot4\cdot3 x^2 + 6\cdot4!) \\
               & = \boxed{\frac{1}{8} (35x^4 - 30x^2 + 3)}
    \end{align*}
    % ----------------------------------------

    \item {}
    จงหาผลเฉลยของสมการ $\displaystyle xy'' + y' - \left(\frac1x - \frac x4\right)y = 0$ \qmark{8}

    % ----------------------------------------
    \bigskip
    Multiply the whole equation by $x$
    $$
        x^2 y'' + xy' - \left(1 - \frac {x^2}4\right) y = 0
    $$

    This is a Bessel's differential equation where
    $$
        x^2 \dv[2]{y}{x} + (1 - 2s) x \dv{y}{x} + \left[a^2r^2x^{2r} + (s^2 - r^2v^2)\right] y =0
    $$

    has general solution in Bessel function form as
    $$
        y(x) = c_1 x^s J_v(ax^r) + c_2 x^s Y_v(ax^r)
    $$

    Use $s = 0, r = 1, a = \frac12, v = 1$; The general solution is
    $$
        \boxed{y = c_1 J_1\left(\frac x2\right) + c_2 Y_1\left(\frac x2\right)}
    $$

    % ----------------------------------------

\end{question}

        \begin{align*}
            f(0)                                   & = \frac{\pi}{4} + \frac{2}{\pi}\sum_{n=1}^{\infty} \frac{1}{(2n-1)^2}                                       \\
            \sum_{n=1}^{\infty} \frac{1}{(2n-1)^2} & = \frac{\pi}{2} \left(f(0) - \frac{\pi}{4}\right)                                                           \\
                                                   & = \frac{\pi}{2} \left(\frac12 \left(\lim_{x\to0^-} f(x) + \lim_{x\to0^+} f(x)\right) - \frac{\pi}{4}\right) \\
                                                   & = \frac{\pi}{2} \left(\frac12 (0 + \pi) - \frac{\pi}{4}\right)                                              \\
                                                   & = \boxed{\frac{\pi^2}{8}}
        \end{align*}
        % ----------------------------------------
    \end{question}

    \item {}
    เส้นลวดมีความยืดหยุ่นเส้นหนึ่งยาว $\pi$ หน่วย ปลายทั้งสองถูกยึดติดแน่นที่จุด $x = 0$ และ $x = \pi$ \\
    เส้นลวดนี้ถูกทำให้เป็นเส้นโค้ง $u(x, 0) = x(\pi - x)$ แล้วปล่อยให้เกิดการสั่นด้วยความเร็ว $u_t(x, 0) = 3$ \\
    จงเขียนปัญหาค่าเริ่มต้นและค่าขอบของโจทย์นี้โดยไม่ต้องหาผลเฉลย (กำหนดให้ $c^2 = 1$) \qmark{3}

    % ----------------------------------------
    $$
        \boxed{
            \begin{array}{c} \displaystyle
                \pdv[2]{u}{t} = \pdv[2]{u}{x}, \quad 0 < x < \pi, \quad t > 0 \\[.5em]
                u(0,t) = 0, \quad u(\pi,t) = 0 , \quad t \geq 0               \\[.5em]
                u(x,0) = x(\pi - x), \quad u_t(x,0) = 3, \quad 0 < x < \pi    \\
            \end{array}
        }
    $$
    % ----------------------------------------

    \item {}
    จงแสดงวิธีหาผลเฉลยของปัญหาค่าเริ่มต้นและค่าขอบซึ่งถูกควบคุมด้วยสมการความร้อนใน $1$ มิติ
    $$
        \pdv{u}{t} = c^2 \pdv[2]{u}{x}, \quad 0 < x < L, \quad t > 0
    $$
    ภายใต้เงื่อนไขขอบ \quad $u(0, t) = 0, \quad u(L, t) = 0, \quad t \geq 0$ \\
    และเงื่อนไขเริ่มต้น \quad $u(x, 0) = f(x), \quad 0 < x < L$ \qmark{12}

    % ----------------------------------------
    $$
        \boxed{
            \begin{array}{c} \displaystyle
                u(x,t) = \sum_{n=1}^\infty A_n e^{-(\frac{n\pi}{L})^2c^2t} \sin\frac{n\pi}{L}x \\[1em]
                \displaystyle A_n = \frac{2}{L} \int_0^L f(x) \sin\frac{n\pi}{L} x \dd{x}, \quad n = 1,2,3,\ldots
            \end{array}
        }
    $$
    % ----------------------------------------

    \item {}
    จงหาผลเฉลยของปัญหาในข้อ 14 เมื่อกำหนดให้ $c^2 = 9, L = 4, f(x) = x + 12$ \qmark{5}

    % ----------------------------------------
    \vspace*{-1em}
    \begin{align*}
        A_n    & = \frac24 \int_0^4 (x + 12) \sin\frac{n\pi}{4}x \dd{x}                                                                                            \\
               & = \frac12 \left[-\frac{4}{n\pi} x\cos\frac{n\pi}{4} x + \frac{16}{n^2\pi^2} \sin\frac{n\pi}{4}x - \frac{48}{n\pi} \cos\frac{n\pi}{4}x \right]_0^4 \\
               & = \frac12 \left(\frac{16(-1)^{n+1}}{n\pi} + \frac{48(-1)^{n+1}}{n\pi} + \frac{48}{n\pi}\right)                                                    \\
               & = \frac{1}{n\pi} \left(32 (-1)^{n+1} + 24\right)                                                                                                  \\
        \intertext{The solution}
        u(x,t) & = \sum_{n=1}^\infty A_n e^{-(\frac{n\pi}{4})^29t} \sin\frac{n\pi}{4}x                                                                             \\
               & = \boxed{\sum_{n=1}^\infty \frac{1}{n\pi} \left(32 (-1)^{n+1} + 24\right) e^{-\frac{9n^2\pi^2}{16}t} \sin\frac{n\pi}{4}x}                         \\
    \end{align*}
    % ----------------------------------------
\end{question}
